import br.edu.ifsp.web.model.Usuario;
import br.edu.ifsp.web.service.*;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;

public class ImportadorConclusoes {

    public static void main(String[] args) {
        if (args.length < 3) {
            System.err.println("Uso: ImportadorConclusoes <caminho_arquivo> <email> <senha>");
            System.exit(1);
        }

        String caminhoArquivo = args[0];
        String email = args[1];
        String senha = args[2];

        AutenticacaoService authService = new AutenticacaoServiceImpl();
        TarefaService tarefaService = new TarefaServiceImpl();

        Usuario usuario;
        try {
            usuario = authService.autenticar(email, senha);
        } catch (AutenticacaoException e) {
            System.err.println("Falha na autenticação: " + e.getMessage());
            System.exit(1);
            return;
        }

        try {
            List<String> linhas = Files.readAllLines(Path.of(caminhoArquivo));
            for (String linha : linhas) {
                if (linha.isBlank()) continue;
                try {
                    int id = Integer.parseInt(linha.trim());
                    tarefaService.concluir(id, usuario);
                    System.out.println("ID " + id + ": concluída");
                } catch (RecursoNaoEncontradoException e) {
                    System.out.println("ID " + linha.trim() + ": não encontrada");
                } catch (RegraNegocioException e) {
                    System.out.println("ID " + linha.trim() + ": já estava concluída");
                } catch (Exception e) {
                    System.out.println("ID " + linha.trim() + ": erro (" + e.getMessage() + ")");
                }
            }
        } catch (Exception e) {
            System.err.println("Erro ao ler arquivo: " + e.getMessage());
            System.exit(1);
        }
    }

package br.edu.ifsp.web.dao;

import br.edu.ifsp.web.model.Tarefa;
import br.edu.ifsp.web.model.Usuario;
import java.sql.*;
import java.util.ArrayList;
import java.util.List;

public class TarefaDAO {

    public List<Tarefa> listarPorUsuario(int usuarioId, Connection conn) throws SQLException {
        String sql = "SELECT id, titulo, concluida FROM tarefas WHERE usuario_id = ?";
        List<Tarefa> lista = new ArrayList<>();
        try (PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setInt(1, usuarioId);
            try (ResultSet rs = stmt.executeQuery()) {
                while (rs.next()) {
                    lista.add(mapear(rs, usuarioId));
                }
            }
        }
        return lista;
    }

    public Tarefa buscarPorIdEUsuario(int id, int usuarioId, Connection conn) throws SQLException {
        String sql = "SELECT id, titulo, concluida FROM tarefas WHERE id = ? AND usuario_id = ?";
        try (PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setInt(1, id);
            stmt.setInt(2, usuarioId);
            try (ResultSet rs = stmt.executeQuery()) {
                if (rs.next()) {
                    return mapear(rs, usuarioId);
                }
                return null;
            }
        }
    }

    public Tarefa inserir(Tarefa tarefa, Connection conn) throws SQLException {
        String sql = "INSERT INTO tarefas (titulo, concluida, usuario_id) VALUES (?, ?, ?)";
        try (PreparedStatement stmt = conn.prepareStatement(sql, Statement.RETURN_GENERATED_KEYS)) {
            stmt.setString(1, tarefa.getTitulo());
            stmt.setBoolean(2, tarefa.isConcluida());
            stmt.setInt(3, tarefa.getUsuario().getId());
            stmt.executeUpdate();
            try (ResultSet keys = stmt.getGeneratedKeys()) {
                if (keys.next()) {
                    tarefa.setId(keys.getInt(1));
                }
            }
            return tarefa;
        }
    }

    public void atualizarTitulo(Tarefa tarefa, Connection conn) throws SQLException {
        String sql = "UPDATE tarefas SET titulo = ? WHERE id = ? AND usuario_id = ?";
        try (PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setString(1, tarefa.getTitulo());
            stmt.setInt(2, tarefa.getId());
            stmt.setInt(3, tarefa.getUsuario().getId());
            stmt.executeUpdate();
        }
    }

    public void marcarComoConcluida(int id, int usuarioId, Connection conn) throws SQLException {
        String sql = "UPDATE tarefas SET concluida = true WHERE id = ? AND usuario_id = ?";
        try (PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setInt(1, id);
            stmt.setInt(2, usuarioId);
            stmt.executeUpdate();
        }
    }

    public boolean excluir(int id, int usuarioId, Connection conn) throws SQLException {
        String sql = "DELETE FROM tarefas WHERE id = ? AND usuario_id = ?";
        try (PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setInt(1, id);
            stmt.setInt(2, usuarioId);
            return stmt.executeUpdate() > 0;
        }
    }

    private Tarefa mapear(ResultSet rs, int usuarioId) throws SQLException {
        Usuario u = new Usuario();
        u.setId(usuarioId);
        return new Tarefa(rs.getInt("id"), rs.getString("titulo"), rs.getBoolean("concluida"), u);
    }

package br.edu.ifsp.web.filter;

import jakarta.servlet.*;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import jakarta.servlet.http.HttpSession;
import java.io.IOException;

public class FiltroAutenticacao implements Filter {

    @Override
    public void doFilter(ServletRequest request, ServletResponse response, FilterChain chain)
            throws IOException, ServletException {
        HttpServletRequest req = (HttpServletRequest) request;
        HttpServletResponse resp = (HttpServletResponse) response;

        // getSession(false) garante que nenhuma sessão seja criada para quem não está autenticado
        HttpSession session = req.getSession(false);

        if (session == null || session.getAttribute("usuarioLogado") == null) {
            resp.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            resp.setContentType("application/json;charset=UTF-8");
            resp.getWriter().write("{\"erro\": \"Usuário não autenticado.\"}");
            return;
        }

        chain.doFilter(request, response);
    }

            throw new AutenticacaoException("E-mail ou senha inválidos.");
        }

        try (Connection conn = connectionFactory.getConnection()) {
            Usuario usuario = usuarioDAO.buscarPorEmail(email.trim(), conn);

            if (usuario == null || !BCrypt.checkpw(senha, usuario.getSenhaHash())) {
                throw new AutenticacaoException("E-mail ou senha inválidos.");
            }

            usuario.setSenhaHash(null);
            return usuario;
        } catch (SQLException e) {
            throw new ErroInternoException("Erro ao autenticar usuário no banco.", e);
        }
    }

import java.util.List;

public class TarefaServiceImpl implements TarefaService {
    private final ConnectionFactory connectionFactory = new ConnectionFactory();
    private final TarefaDAO tarefaDAO = new TarefaDAO();

    @Override
    public List<Tarefa> listar(Usuario usuario) {
        try (Connection conn = connectionFactory.getConnection()) {
            return tarefaDAO.listarPorUsuario(usuario.getId(), conn);
        } catch (SQLException e) {
            throw new ErroInternoException("Erro ao listar tarefas.", e);
        }
    }

    @Override
    public Tarefa buscar(int id, Usuario usuario) {
        try (Connection conn = connectionFactory.getConnection()) {
            Tarefa tarefa = tarefaDAO.buscarPorIdEUsuario(id, usuario.getId(), conn);
            if (tarefa == null) {
                throw new RecursoNaoEncontradoException("Tarefa não encontrada.");
            }
            return tarefa;
        } catch (SQLException e) {
            throw new ErroInternoException("Erro ao buscar tarefa.", e);
        }
    }

    @Override
    public Tarefa criar(String titulo, Usuario usuario) {
        Tarefa nova = new Tarefa(0, titulo, false, usuario);
        try (Connection conn = connectionFactory.getConnection()) {
            conn.setAutoCommit(false);
            try {
                Tarefa criada = tarefaDAO.inserir(nova, conn);
                conn.commit();
                return criada;
            } catch (Exception e) {
                conn.rollback();
                throw e;
            }
        } catch (SQLException e) {
            throw new ErroInternoException("Erro ao criar tarefa.", e);
        }
    }

    @Override
    public Tarefa atualizar(int id, String novoTitulo, Usuario usuario) {
        try (Connection conn = connectionFactory.getConnection()) {
            conn.setAutoCommit(false);
            try {
                Tarefa tarefa = tarefaDAO.buscarPorIdEUsuario(id, usuario.getId(), conn);
                if (tarefa == null) {
                    throw new RecursoNaoEncontradoException("Tarefa não encontrada.");
                }
                tarefa.setTitulo(novoTitulo);
                tarefaDAO.atualizarTitulo(tarefa, conn);
                conn.commit();
                return tarefa;
            } catch (Exception e) {
                conn.rollback();
                throw e;
            }
        } catch (SQLException e) {
            throw new ErroInternoException("Erro ao atualizar tarefa.", e);
        }
    }

    @Override
    public void concluir(int id, Usuario usuario) {
        try (Connection conn = connectionFactory.getConnection()) {
            conn.setAutoCommit(false);
            try {
                Tarefa tarefa = tarefaDAO.buscarPorIdEUsuario(id, usuario.getId(), conn);
                if (tarefa == null) {
                    throw new RecursoNaoEncontradoException("Tarefa não encontrada.");
                }
                tarefa.concluir();
                tarefaDAO.marcarComoConcluida(tarefa.getId(), usuario.getId(), conn);
                conn.commit();
            } catch (Exception e) {
                conn.rollback();
                throw e;
            }
        } catch (SQLException e) {
            throw new ErroInternoException("Erro ao concluir tarefa.", e);
        }
    }

        try (Connection conn = connectionFactory.getConnection()) {
            conn.setAutoCommit(false);

                boolean excluiu = tarefaDAO.excluir(id, usuario.getId(), conn);
                if (!excluiu) {
                    throw new RecursoNaoEncontradoException("Tarefa não encontrada.");
                }
                conn.commit();
            } catch (Exception e) {
                conn.rollback();
                throw e;
            }
        } catch (SQLException e) {
            throw new ErroInternoException("Erro ao excluir tarefa.", e);
        }
    }

package br.edu.ifsp.web.servlet;

import br.edu.ifsp.web.listener.SessaoListener;
import br.edu.ifsp.web.model.Usuario;
import br.edu.ifsp.web.service.AutenticacaoException;
import br.edu.ifsp.web.service.AutenticacaoService;
import br.edu.ifsp.web.service.AutenticacaoServiceImpl;
import com.google.gson.Gson;
import com.google.gson.JsonObject;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.*;
import java.io.IOException;
import java.util.Map;

@WebServlet("/auth/*")
public class AuthServlet extends HttpServlet {
    private final AutenticacaoService authService = new AutenticacaoServiceImpl();
    private final Gson gson = new Gson();

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws IOException {
        String path = req.getPathInfo();
        resp.setContentType("application/json;charset=UTF-8");

        if ("/login".equals(path)) {
            try {
                JsonObject body = gson.fromJson(req.getReader(), JsonObject.class);
                String email = (body != null && body.has("email")) ? body.get("email").getAsString() : null;
                String senha = (body != null && body.has("senha")) ? body.get("senha").getAsString() : null;

                Usuario usuario = authService.autenticar(email, senha);

                // Se já havia uma sessão anterior, invalida antes de criar a nova
                HttpSession sessaoAntiga = req.getSession(false);
                if (sessaoAntiga != null) {
                    sessaoAntiga.invalidate();
                }

                HttpSession novaSessao = req.getSession(true);
                req.changeSessionId(); // Garante que o identificador da sessão mude após login bem-sucedido
                novaSessao.setAttribute("usuarioLogado", usuario);

                resp.setStatus(HttpServletResponse.SC_OK);
                resp.getWriter().write(gson.toJson(usuario)); // senhaHash é transient, nunca vai no JSON
            } catch (AutenticacaoException e) {
                resp.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
                resp.getWriter().write(gson.toJson(Map.of("erro", e.getMessage())));
            } catch (Exception e) {
                resp.setStatus(HttpServletResponse.SC_INTERNAL_SERVER_ERROR);
                resp.getWriter().write(gson.toJson(Map.of("erro", "Erro interno ao autenticar.")));
            }
        } else if ("/logout".equals(path)) {
            HttpSession session = req.getSession(false);
            if (session != null) {
                session.invalidate();
            }
            resp.setStatus(HttpServletResponse.SC_NO_CONTENT);
        } else {
            resp.setStatus(HttpServletResponse.SC_NOT_FOUND);
            resp.getWriter().write(gson.toJson(Map.of("erro", "Rota não encontrada.")));
        }
    }

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp) throws IOException {
        resp.setContentType("application/json;charset=UTF-8");

        if ("/me".equals(req.getPathInfo())) {
            HttpSession session = req.getSession(false);
            Usuario usuario = (session != null) ? (Usuario) session.getAttribute("usuarioLogado") : null;

            if (usuario == null) {
                resp.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
                resp.getWriter().write(gson.toJson(Map.of("erro", "Usuário não autenticado.")));
                return;
            }

            resp.setStatus(HttpServletResponse.SC_OK);
            resp.getWriter().write(gson.toJson(Map.of(
                    "usuario", usuario,
                    "sessoesAtivas", SessaoListener.getSessoesAtivas()
            )));
        } else {
            resp.setStatus(HttpServletResponse.SC_NOT_FOUND);
            resp.getWriter().write(gson.toJson(Map.of("erro", "Rota não encontrada.")));
        }
    }

package br.edu.ifsp.web.servlet;

import br.edu.ifsp.web.command.*;
import br.edu.ifsp.web.service.*;
import com.google.gson.Gson;
import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import java.io.IOException;
import java.util.Map;

@WebServlet("/tarefas/*")
public class TaskServlet extends HttpServlet {
    private final TarefaService tarefaService = new TarefaServiceImpl();
    private final Gson gson = new Gson();

    @Override
    protected void service(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        resp.setContentType("application/json;charset=UTF-8");
        try {
            Command cmd = resolverCommand(req);
            cmd.execute(req, resp, tarefaService);
        } catch (DadosInvalidosException | NumberFormatException e) {
            enviarErro(resp, HttpServletResponse.SC_BAD_REQUEST, e.getMessage());
        } catch (RecursoNaoEncontradoException e) {
            enviarErro(resp, HttpServletResponse.SC_NOT_FOUND, e.getMessage());
        } catch (RegraNegocioException e) {
            enviarErro(resp, HttpServletResponse.SC_CONFLICT, e.getMessage());
        } catch (Exception e) {
            enviarErro(resp, HttpServletResponse.SC_INTERNAL_SERVER_ERROR, "Erro interno no servidor.");
        }
    }

    private Command resolverCommand(HttpServletRequest req) {
        String method = req.getMethod();
        String path = req.getPathInfo();
        boolean temId = path != null && path.length() > 1;

        if ("GET".equals(method) && !temId) return new ListarTarefasCommand();
        if ("GET".equals(method) && temId)  return new BuscarTarefaCommand();
        if ("POST".equals(method) && !temId) return new CriarTarefaCommand();
        if ("PUT".equals(method) && temId)  return new AtualizarTarefaCommand();
        if ("PATCH".equals(method) && temId) return new ConcluirTarefaCommand();
        if ("DELETE".equals(method) && temId) return new ExcluirTarefaCommand();
        throw new RecursoNaoEncontradoException("Rota não encontrada.");
    }

    private void enviarErro(HttpServletResponse resp, int status, String msg) throws IOException {
        resp.setStatus(status);
        resp.getWriter().write(gson.toJson(Map.of("erro", msg != null ? msg : "Erro na requisição.")));
    }
